#!/bin/sh
# Recorrido de diseno 002 Integracion paso 6 sobre un commit: mensaje T-NNN (FFF): y carpeta con los 3 archivos
c=$1
msg=$(git log -1 --format=%s $c)
t=$(echo "$msg" | sed -nE 's/^(T-[0-9]+) \(([0-9]+)\): .*/\1 \2/p')
[ -z "$t" ] && { echo "FALLA $c: mensaje sin forma T-NNN (FFF):"; exit 1; }
set -- $t; id=$1; f=$2
d=$(git ls-tree -r --name-only $c | grep -E "^specs/$f-[^/]*/evidencias/$id/" )
ok=0
for a in verificar.md revision.md aprobacion.md; do echo "$d" | grep -q "/$a$" || { echo "FALLA $c: falta $a"; ok=1; }; done
[ $ok = 0 ] && echo "OK $c"; exit $ok
