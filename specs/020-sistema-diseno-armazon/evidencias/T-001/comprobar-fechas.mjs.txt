import fs from 'node:fs';
const [a, b, limite] = process.argv.slice(2);
const A = JSON.parse(fs.readFileSync(a)).packages;
const B = JSON.parse(fs.readFileSync(b)).packages;
const nuevos = [];
for (const [k, v] of Object.entries(B)) {
  if (!k) continue;
  if (!A[k] || A[k].version !== v.version) {
    nuevos.push([k.replace(/^.*node_modules\//, ''), v.version, v.hasInstallScript ? 'SCRIPT' : '']);
  }
}
const uniq = [...new Map(nuevos.map((n) => [n[0] + '@' + n[1], n])).values()];
let mal = 0;
for (const [n, v, s] of uniq) {
  const r = await fetch('https://registry.npmjs.org/' + n.replace('/', '%2F'));
  const j = await r.json();
  const t = j.time?.[v];
  const ok = t && t <= limite;
  if (!ok) mal++;
  console.log(`${ok ? 'ok ' : 'MAL'} ${n}@${v} ${t} ${s}`);
}
console.log('total', uniq.length, 'fuera de límite', mal);
