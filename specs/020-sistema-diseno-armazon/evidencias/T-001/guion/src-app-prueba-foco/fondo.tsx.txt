"use client";

// Página desechable de T-001 (020). Datos ficticios. Fondo común: tabla, formulario y gráfico.
import { useEffect, type ReactNode } from "react";
import { Bar, BarChart, XAxis, YAxis } from "recharts";
import "./prueba.css";

const FILAS = [
  { id: 1, nombre: "Cliente Ficticio Uno", total: 12 },
  { id: 2, nombre: "Cliente Ficticio Dos", total: 7 },
  { id: 3, nombre: "Cliente Ficticio Tres", total: 21 },
  { id: 4, nombre: "Cliente Ficticio Cuatro", total: 3 },
];

export function Fondo({
  titulo,
  acciones,
  confirmaciones,
}: {
  titulo: string;
  acciones: ReactNode;
  confirmaciones: number;
}) {
  useEffect(() => {
    document.body.dataset.hidratado = "1";
  }, []);
  return (
    <div className="pagina">
      <h1>{titulo}</h1>
      <p>
        Confirmaciones registradas: <output data-testid="confirmaciones">{confirmaciones}</output>
      </p>
      <section aria-label="Acciones de la prueba" className="acciones">
        {acciones}
      </section>

      <table>
        <caption>Tabla ficticia de clientes</caption>
        <thead>
          <tr>
            <th scope="col">Nombre</th>
            <th scope="col">Total</th>
            <th scope="col">Acción</th>
          </tr>
        </thead>
        <tbody>
          {FILAS.map((f) => (
            <tr key={f.id}>
              <th scope="row">{f.nombre}</th>
              <td>{f.total}</td>
              <td>
                <button type="button" id={`fila-${f.id}`}>
                  Ver {f.nombre}
                </button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>

      <form onSubmit={(e) => e.preventDefault()} aria-label="Formulario ficticio">
        <label htmlFor="campo-nombre">Nombre de prueba</label>
        <input id="campo-nombre" name="nombre" type="text" autoComplete="off" />
        <label htmlFor="campo-tipo">Tipo</label>
        <select id="campo-tipo" name="tipo" defaultValue="a">
          <option value="a">Tipo A</option>
          <option value="b">Tipo B</option>
        </select>
        <label>
          <input type="checkbox" name="activo" /> Activo
        </label>
        <button type="submit">Guardar formulario ficticio</button>
      </form>

      <figure>
        <div role="img" aria-label="Gráfico de barras de totales ficticios por cliente">
          <BarChart width={480} height={200} data={FILAS} accessibilityLayer={false}>
            <XAxis dataKey="id" />
            <YAxis />
            <Bar dataKey="total" fill="#1f4e79" />
          </BarChart>
        </div>
        <figcaption>Totales ficticios; los mismos valores están en la tabla.</figcaption>
      </figure>

      <button type="button" id="fuera-final">
        Botón final de la página
      </button>
    </div>
  );
}
